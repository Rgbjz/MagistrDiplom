# Plan 1 — Scaffold + Prisma (Phases 0–1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Prepare the repo for the rewrite, scaffold empty NestJS and Next.js+TS apps, and stand up the Prisma schema (16 tables) with a working migration and seed.

**Architecture:** Move existing apps into `legacy/`. Create fresh `backend/` (NestJS) and `frontend/` (Next.js App Router + TS). Author `prisma/schema.prisma` by hand from the legacy Sequelize models (13 tables) plus 3 chat tables, migrate from scratch, and seed dev data.

**Tech Stack:** Node v24 (nvm), NestJS 11, Prisma 6 + PostgreSQL, Next.js 15 (App Router) + TypeScript.

> **Note on tests:** Automated tests are out of scope per the design spec. Verification steps use typecheck/build/migrate/seed/curl instead of unit tests.

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ===== Enums =====
enum UserRole {
  STUDENT
  TEACHER
  ADMIN
}

enum EnrollmentStatus {
  PENDING
  ACTIVE
  REJECTED
  COMPLETED
}

enum LessonProgressStatus {
  NOT_STARTED
  IN_PROGRESS
  COMPLETED
}

enum QuestionType {
  SINGLE
  MULTIPLE
}

enum ConversationType {
  COURSE
  DIRECT
}

// ===== Core =====
model User {
  id        Int      @id @default(autoincrement())
  email     String   @unique
  password  String
  role      UserRole @default(STUDENT)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  profile          UserProfile?
  sessions         Session[]
  courses          Course[]         @relation("TeacherCourses")
  enrollments      Enrollment[]
  lessonProgresses LessonProgress[]
  testResults      TestResult[]
  userAnswers      UserAnswer[]
  messages         Message[]
  participations   ConversationParticipant[]
}

model UserProfile {
  id        Int      @id @default(autoincrement())
  firstName String?
  lastName  String?
  avatarUrl String?
  bio       String?
  userId    Int      @unique
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model Session {
  id               Int      @id @default(autoincrement())
  userId           Int
  refreshTokenHash String
  userAgent        String?
  ip               String?
  expiresAt        DateTime
  createdAt        DateTime @default(now())
  updatedAt        DateTime @updatedAt

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)
}

model Course {
  id          Int      @id @default(autoincrement())
  title       String
  description String?
  imageUrl    String?
  teacherId   Int
  createdAt   DateTime @default(now())
  updatedAt   DateTime @updatedAt

  teacher       User           @relation("TeacherCourses", fields: [teacherId], references: [id])
  sections      Section[]
  enrollments   Enrollment[]
  conversations Conversation[]
}

model Section {
  id        Int      @id @default(autoincrement())
  title     String
  order     Int
  courseId  Int
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  course  Course   @relation(fields: [courseId], references: [id], onDelete: Cascade)
  lessons Lesson[]
}

model Lesson {
  id        Int      @id @default(autoincrement())
  title     String
  content   String?
  videoUrl  String?
  order     Int
  sectionId Int
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  section    Section          @relation(fields: [sectionId], references: [id], onDelete: Cascade)
  progresses LessonProgress[]
  test       Test?
}

model Test {
  id           Int      @id @default(autoincrement())
  lessonId     Int      @unique
  title        String   @default("Test")
  timeLimit    Int
  passingScore Int      @default(60)
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  lesson    Lesson       @relation(fields: [lessonId], references: [id], onDelete: Cascade)
  questions Question[]
  results   TestResult[]
}

model Question {
  id         Int          @id @default(autoincrement())
  testId     Int
  text       String
  type       QuestionType @default(SINGLE)
  difficulty Int          @default(1)
  createdAt  DateTime     @default(now())
  updatedAt  DateTime     @updatedAt

  test        Test         @relation(fields: [testId], references: [id], onDelete: Cascade)
  answers     Answer[]
  userAnswers UserAnswer[]
}

model Answer {
  id         Int      @id @default(autoincrement())
  questionId Int
  text       String
  isCorrect  Boolean  @default(false)
  createdAt  DateTime @default(now())
  updatedAt  DateTime @updatedAt

  question    Question     @relation(fields: [questionId], references: [id], onDelete: Cascade)
  userAnswers UserAnswer[]
}

model Enrollment {
  id          Int              @id @default(autoincrement())
  userId      Int
  courseId    Int
  status      EnrollmentStatus @default(PENDING)
  enrolledAt  DateTime         @default(now())
  completedAt DateTime?
  createdAt   DateTime         @default(now())
  updatedAt   DateTime         @updatedAt

  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)
  course Course @relation(fields: [courseId], references: [id], onDelete: Cascade)

  @@unique([userId, courseId])
}

model LessonProgress {
  id          Int                  @id @default(autoincrement())
  userId      Int
  lessonId    Int
  status      LessonProgressStatus @default(NOT_STARTED)
  startedAt   DateTime?
  completedAt DateTime?
  createdAt   DateTime             @default(now())
  updatedAt   DateTime             @updatedAt

  user   User   @relation(fields: [userId], references: [id], onDelete: Cascade)
  lesson Lesson @relation(fields: [lessonId], references: [id], onDelete: Cascade)

  @@unique([userId, lessonId])
}

model TestResult {
  id         Int       @id @default(autoincrement())
  userId     Int
  testId     Int
  attempt    Int
  startedAt  DateTime
  finishedAt DateTime?
  score      Int?
  passed     Boolean?
  createdAt  DateTime  @default(now())
  updatedAt  DateTime  @updatedAt

  user    User         @relation(fields: [userId], references: [id], onDelete: Cascade)
  test    Test         @relation(fields: [testId], references: [id], onDelete: Cascade)
  answers UserAnswer[]
}

model UserAnswer {
  id           Int      @id @default(autoincrement())
  userId       Int
  questionId   Int
  answerId     Int
  testResultId Int
  createdAt    DateTime @default(now())
  updatedAt    DateTime @updatedAt

  user       User       @relation(fields: [userId], references: [id], onDelete: Cascade)
  question   Question   @relation(fields: [questionId], references: [id], onDelete: Cascade)
  answer     Answer     @relation(fields: [answerId], references: [id], onDelete: Cascade)
  testResult TestResult @relation(fields: [testResultId], references: [id], onDelete: Cascade)
}

// ===== Chat =====
model Conversation {
  id        Int              @id @default(autoincrement())
  type      ConversationType
  courseId  Int?
  createdAt DateTime         @default(now())
  updatedAt DateTime         @updatedAt

  course       Course?                   @relation(fields: [courseId], references: [id], onDelete: Cascade)
  participants ConversationParticipant[]
  messages     Message[]
}

model ConversationParticipant {
  id             Int      @id @default(autoincrement())
  conversationId Int
  userId         Int
  createdAt      DateTime @default(now())

  conversation Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
  user         User         @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([conversationId, userId])
}

model Message {
  id             Int       @id @default(autoincrement())
  conversationId Int
  senderId       Int
  content        String
  readAt         DateTime?
  createdAt      DateTime  @default(now())

  conversation Conversation @relation(fields: [conversationId], references: [id], onDelete: Cascade)
  sender       User         @relation(fields: [senderId], references: [id], onDelete: Cascade)
}
```

## Task 7: Write the seed script

**Files:**
- Create: `backend/prisma/seed.ts`

- [ ] **Step 1: Write `backend/prisma/seed.ts`**

```typescript
import { PrismaClient } from '@prisma/client'
import bcrypt from 'bcrypt'

const prisma = new PrismaClient()

async function main() {
  const passwordHash = await bcrypt.hash('password123', 10)

  // Teacher
  const teacher = await prisma.user.create({
    data: {
      email: 'teacher@example.com',
      password: passwordHash,
      role: 'TEACHER',
      profile: { create: { firstName: 'Tessa', lastName: 'Teacher' } },
    },
  })

  // Students
  const student1 = await prisma.user.create({
    data: {
      email: 'student1@example.com',
      password: passwordHash,
      role: 'STUDENT',
      profile: { create: { firstName: 'Sam', lastName: 'Student' } },
    },
  })

  const student2 = await prisma.user.create({
    data: {
      email: 'student2@example.com',
      password: passwordHash,
      role: 'STUDENT',
      profile: { create: { firstName: 'Sara', lastName: 'Scholar' } },
    },
  })

  // Course with section -> lesson -> test -> questions -> answers
  const course = await prisma.course.create({
    data: {
      title: 'Intro to TypeScript',
      description: 'A starter course.',
      teacherId: teacher.id,
      sections: {
        create: {
          title: 'Getting Started',
          order: 1,
          lessons: {
            create: {
              title: 'What is TypeScript?',
              content: 'TypeScript is a typed superset of JavaScript.',
              order: 1,
              test: {
                create: {
                  title: 'Lesson 1 Quiz',
                  timeLimit: 10,
                  passingScore: 60,
                  questions: {
                    create: [
                      {
                        text: 'TypeScript compiles to?',
                        type: 'SINGLE',
                        difficulty: 1,
                        answers: {
                          create: [
                            { text: 'JavaScript', isCorrect: true },
                            { text: 'Python', isCorrect: false },
                            { text: 'WebAssembly', isCorrect: false },
                          ],
                        },
                      },
                    ],
                  },
                },
              },
            },
          },
        },
      },
    },
  })

  // Enrollments: student1 ACTIVE, student2 PENDING
  await prisma.enrollment.createMany({
    data: [
      { userId: student1.id, courseId: course.id, status: 'ACTIVE' },
      { userId: student2.id, courseId: course.id, status: 'PENDING' },
    ],
  })

  console.log('Seed complete:', {
    teacher: teacher.email,
    students: [student1.email, student2.email],
    course: course.title,
  })
}

main()
  .then(async () => {
    await prisma.$disconnect()
  })
  .catch(async (e) => {
    console.error(e)
    await prisma.$disconnect()
    process.exit(1)
  })
```
