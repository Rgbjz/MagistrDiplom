# Plan 4 — Chat (Socket.io) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add realtime chat — course rooms (teacher + ACTIVE students) and 1-on-1 direct messages — via a NestJS Socket.io gateway and a `/chat` UI. This is the one new feature beyond the legacy app.

**Architecture:** Backend `ChatModule` = `ChatService` (Prisma over the existing `Conversation`/`ConversationParticipant`/`Message` tables) + `ChatController` (REST: list conversations, message history, open course/direct conversation) + `ChatGateway` (Socket.io, authenticates the handshake JWT, joins `conversation:<id>` rooms, broadcasts new messages). Frontend `/chat` page uses TanStack Query for history and a socket.io-client (connected with the session access token) for live `message:new` events appended to the query cache. Entry points: a "Course chat" button on the CoursePage and a "Message" button (teacher→student) on the management page.

**Tech Stack:** NestJS 11 + `@nestjs/websockets` + `@nestjs/platform-socket.io` + `socket.io`; frontend `socket.io-client`. Builds on all prior plans.

> **Tests out of scope.** Verification: build, boot, REST checks, and a real two-client socket message exchange via a node script.

> **Env note:** Node not on default PATH — `export PATH="$HOME/.nvm/versions/node/v24.15.0/bin:$PATH"`. Backend :5000 (`lsof -ti tcp:5000 | xargs kill -9` first), frontend :3000. Postgres must run.

> **Auth:** the access-token JWT payload is `{ id, role, email }`, signed with `JWT_ACCESS_SECRET`. The gateway verifies `socket.handshake.auth.token`. The frontend gets the token from the Auth.js session (`session.accessToken`).

> **Existing Prisma models (Plan 1):** `Conversation { id, type: ConversationType(COURSE|DIRECT), courseId?, createdAt, updatedAt, participants[], messages[] }`, `ConversationParticipant { id, conversationId, userId, @@unique([conversationId, userId]) }` (selector `conversationId_userId`), `Message { id, conversationId, senderId, content, readAt?, createdAt, sender }`.

```typescript
import {
  BadRequestException,
  ForbiddenException,
  Injectable,
  NotFoundException,
} from '@nestjs/common';
import { PrismaService } from '../../prisma/prisma.service';
import type { AuthUser } from '../../common/decorators/current-user.decorator';

const senderSelect = {
  id: true,
  email: true,
  profile: { select: { firstName: true, lastName: true } },
};

@Injectable()
export class ChatService {
  constructor(private prisma: PrismaService) {}

  private async courseConversationAccess(courseId: number, user: AuthUser) {
    const course = await this.prisma.course.findUnique({
      where: { id: courseId },
    });
    if (!course) throw new NotFoundException('Course not found');
    if (user.role === 'ADMIN' || course.teacherId === user.id) return;
    const enrollment = await this.prisma.enrollment.findFirst({
      where: { userId: user.id, courseId, status: 'ACTIVE' },
    });
    if (!enrollment) throw new ForbiddenException('Not enrolled in this course');
  }

  /** Throws unless the user may access the conversation; lazily adds them to a COURSE room they can access. */
  async assertAccess(conversationId: number, user: AuthUser) {
    const participant = await this.prisma.conversationParticipant.findUnique({
      where: { conversationId_userId: { conversationId, userId: user.id } },
    });
    if (participant) return;

    const conversation = await this.prisma.conversation.findUnique({
      where: { id: conversationId },
    });
    if (!conversation) throw new NotFoundException('Conversation not found');

    if (conversation.type === 'COURSE' && conversation.courseId) {
      await this.courseConversationAccess(conversation.courseId, user);
      await this.prisma.conversationParticipant.create({
        data: { conversationId, userId: user.id },
      });
      return;
    }
    throw new ForbiddenException('Not a participant');
  }

  async getCourseConversation(courseId: number, user: AuthUser) {
    await this.courseConversationAccess(courseId, user);
    let conversation = await this.prisma.conversation.findFirst({
      where: { type: 'COURSE', courseId },
    });
    if (!conversation) {
      conversation = await this.prisma.conversation.create({
        data: { type: 'COURSE', courseId },
      });
    }
    await this.prisma.conversationParticipant.upsert({
      where: {
        conversationId_userId: {
          conversationId: conversation.id,
          userId: user.id,
        },
      },
      create: { conversationId: conversation.id, userId: user.id },
      update: {},
    });
    return this.dto(conversation.id, user.id);
  }

  async getOrCreateDirect(userId: number, otherUserId: number) {
    if (userId === otherUserId) {
      throw new BadRequestException('Cannot message yourself');
    }
    const other = await this.prisma.user.findUnique({
      where: { id: otherUserId },
    });
    if (!other) throw new NotFoundException('User not found');

    let conversation = await this.prisma.conversation.findFirst({
      where: {
        type: 'DIRECT',
        AND: [
          { participants: { some: { userId } } },
          { participants: { some: { userId: otherUserId } } },
        ],
      },
    });
    if (!conversation) {
      conversation = await this.prisma.conversation.create({
        data: {
          type: 'DIRECT',
          participants: { create: [{ userId }, { userId: otherUserId }] },
        },
      });
    }
    return this.dto(conversation.id, userId);
  }

  async listConversations(userId: number) {
    const conversations = await this.prisma.conversation.findMany({
      where: { participants: { some: { userId } } },
      include: {
        course: { select: { id: true, title: true } },
        participants: { include: { user: { select: senderSelect } } },
        messages: { orderBy: { id: 'desc' }, take: 1 },
      },
      orderBy: { updatedAt: 'desc' },
    });
    return conversations.map((c) => this.toDto(c, userId));
  }

  async getMessages(conversationId: number, user: AuthUser, cursor?: number) {
    await this.assertAccess(conversationId, user);
    const messages = await this.prisma.message.findMany({
      where: { conversationId },
      orderBy: { id: 'desc' },
      take: 30,
      ...(cursor ? { cursor: { id: cursor }, skip: 1 } : {}),
      include: { sender: { select: senderSelect } },
    });
    return messages.reverse();
  }

  async createMessage(conversationId: number, sender: AuthUser, content: string) {
    const text = content?.trim();
    if (!text) throw new BadRequestException('Empty message');
    await this.assertAccess(conversationId, sender);
    const message = await this.prisma.message.create({
      data: { conversationId, senderId: sender.id, content: text },
      include: { sender: { select: senderSelect } },
    });
    await this.prisma.conversation.update({
      where: { id: conversationId },
      data: { updatedAt: new Date() },
    });
    return message;
  }

  /** Participant user ids (for targeted gateway broadcast if needed). */
  async participantIds(conversationId: number): Promise<number[]> {
    const ps = await this.prisma.conversationParticipant.findMany({
      where: { conversationId },
      select: { userId: true },
    });
    return ps.map((p) => p.userId);
  }

  private async dto(conversationId: number, viewerId: number) {
    const c = await this.prisma.conversation.findUnique({
      where: { id: conversationId },
      include: {
        course: { select: { id: true, title: true } },
        participants: { include: { user: { select: senderSelect } } },
        messages: { orderBy: { id: 'desc' }, take: 1 },
      },
    });
    return this.toDto(c!, viewerId);
  }

  private toDto(
    c: {
      id: number;
      type: string;
      courseId: number | null;
      course: { id: number; title: string } | null;
      participants: {
        user: { id: number; email: string; profile: { firstName: string | null; lastName: string | null } | null };
      }[];
      messages: { id: number; content: string; createdAt: Date }[];
    },
    viewerId: number,
  ) {
    let title: string;
    if (c.type === 'COURSE') {
      title = c.course ? `Course: ${c.course.title}` : 'Course chat';
    } else {
      const other = c.participants.find((p) => p.user.id !== viewerId)?.user;
      const prof = other?.profile;
      title =
        prof && (prof.firstName || prof.lastName)
          ? `${prof.firstName ?? ''} ${prof.lastName ?? ''}`.trim()
          : (other?.email ?? 'Direct message');
    }
    const last = c.messages[0];
    return {
      id: c.id,
      type: c.type,
      courseId: c.courseId,
      title,
      lastMessage: last
        ? { content: last.content, createdAt: last.createdAt }
        : null,
    };
  }
}
```

  Body,
  Controller,
  Get,
  Param,
  ParseIntPipe,
  Post,
  Query,
  UseGuards,

import { JwtAuthGuard } from '../../common/auth/jwt-auth.guard';
import { CurrentUser } from '../../common/decorators/current-user.decorator';
import type { AuthUser } from '../../common/decorators/current-user.decorator';
import { ChatService } from './chat.service';
import { DirectDto } from './dto/direct.dto';

@Controller('chat')
@UseGuards(JwtAuthGuard)
export class ChatController {
  constructor(private chat: ChatService) {}

  @Get('conversations')
  list(@CurrentUser('id') userId: number) {
    return this.chat.listConversations(userId);
  }

  @Get('conversations/:id/messages')
  messages(
    @Param('id', ParseIntPipe) id: number,
    @CurrentUser() user: AuthUser,
    @Query('cursor') cursor?: string,

    return this.chat.getMessages(id, user, cursor ? Number(cursor) : undefined);

  @Post('conversations/direct')

    return this.chat.getOrCreateDirect(userId, dto.userId);
  }

  @Get('courses/:courseId/conversation')
  courseConversation(
    @Param('courseId', ParseIntPipe) courseId: number,
    @CurrentUser() user: AuthUser,
  ) {
    return this.chat.getCourseConversation(courseId, user);
  }
}

```typescript
import {
  ConnectedSocket,
  MessageBody,
  OnGatewayConnection,
  SubscribeMessage,
  WebSocketGateway,
  WebSocketServer,
} from '@nestjs/websockets';
import { JwtService } from '@nestjs/jwt';
import { ConfigService } from '@nestjs/config';
import type { Server, Socket } from 'socket.io';
import { ChatService } from './chat.service';
import type { AuthUser } from '../../common/decorators/current-user.decorator';

interface JwtPayload {
  id: number;
  role: AuthUser['role'];
  email?: string;
}

@WebSocketGateway({
  cors: {
    origin: process.env.FRONTEND_URL ?? 'http://localhost:3000',
    credentials: true,
  },
})
export class ChatGateway implements OnGatewayConnection {
  @WebSocketServer()
  server: Server;

  constructor(
    private chat: ChatService,
    private jwt: JwtService,
    private config: ConfigService,
  ) {}

  handleConnection(client: Socket) {
    const token = client.handshake.auth?.token as string | undefined;
    if (!token) {
      client.disconnect();
      return;
    }
    try {
      const payload = this.jwt.verify<JwtPayload>(token, {
        secret: this.config.get<string>('JWT_ACCESS_SECRET'),
      });
      client.data.user = {
        id: payload.id,
        role: payload.role,
        email: payload.email,
      } satisfies AuthUser;
    } catch {
      client.disconnect();
    }
  }

  private user(client: Socket): AuthUser {
    return client.data.user as AuthUser;
  }

  @SubscribeMessage('conversation:join')
  async join(
    @ConnectedSocket() client: Socket,
    @MessageBody() body: { conversationId: number },
  ) {
    try {
      await this.chat.assertAccess(body.conversationId, this.user(client));
      client.join(`conversation:${body.conversationId}`);
      return { ok: true };
    } catch {
      return { ok: false };
    }
  }

  @SubscribeMessage('conversation:leave')
  leave(
    @ConnectedSocket() client: Socket,
    @MessageBody() body: { conversationId: number },
  ) {
    client.leave(`conversation:${body.conversationId}`);
    return { ok: true };
  }

  @SubscribeMessage('message:send')
  async send(
    @ConnectedSocket() client: Socket,
    @MessageBody() body: { conversationId: number; content: string },
  ) {
    const message = await this.chat.createMessage(
      body.conversationId,
      this.user(client),
      body.content,
    );
    this.server
      .to(`conversation:${body.conversationId}`)
      .emit('message:new', message);
    return { ok: true };
  }
}
```

'use client';

import { io, type Socket } from 'socket.io-client';

const BASE = (
  process.env.NEXT_PUBLIC_API_URL ?? 'http://localhost:5000/api'
).replace(/\/api$/, '');

let socket: Socket | null = null;
let socketToken: string | null = null;

export function getSocket(token: string): Socket {
  if (socket && socketToken === token) return socket;
  if (socket) socket.disconnect();
  socket = io(BASE, { auth: { token }, transports: ['websocket'] });
  socketToken = token;
  return socket;
}

```typescript
'use client';

import { useEffect, useRef, useState } from 'react';
import { useQuery, useQueryClient } from '@tanstack/react-query';
import { useSession } from 'next-auth/react';
import { clientApi } from '../../lib/api/client';
import { getSocket } from '../../lib/socket';
import type { ChatMessage, SessionUser } from '../../lib/types';
import styles from '../../app/(app)/chat/chat.module.scss';

export function ChatThread({ conversationId }: { conversationId: number }) {
  const { data: session } = useSession();
  const queryClient = useQueryClient();
  const [text, setText] = useState('');
  const bottomRef = useRef<HTMLDivElement>(null);
  const token = session?.accessToken;
  const me = session?.user as SessionUser | undefined;

  const { data: messages } = useQuery<ChatMessage[]>({
    queryKey: ['messages', conversationId],
    queryFn: () =>
      clientApi<ChatMessage[]>(`/chat/conversations/${conversationId}/messages`),
  });

  useEffect(() => {
    if (!token) return;
    const socket = getSocket(token);
    socket.emit('conversation:join', { conversationId });

    const onNew = (msg: ChatMessage) => {
      if (msg.conversationId !== conversationId) return;
      queryClient.setQueryData<ChatMessage[]>(
        ['messages', conversationId],
        (prev) => {
          if (!prev) return [msg];
          if (prev.some((m) => m.id === msg.id)) return prev;
          return [...prev, msg];
        },
      );
    };
    socket.on('message:new', onNew);
    return () => {
      socket.off('message:new', onNew);
      socket.emit('conversation:leave', { conversationId });
    };
  }, [conversationId, token, queryClient]);

  useEffect(() => {
    bottomRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);

  function send() {
    if (!text.trim() || !token) return;
    getSocket(token).emit('message:send', {
      conversationId,
      content: text.trim(),
    });
    setText('');
  }

  return (
    <div className={styles.thread}>
      <div className={styles.messages}>
        {(messages ?? []).map((m) => (
          <div
            key={m.id}
            className={`${styles.message} ${m.senderId === me?.id ? styles.mine : ''}`}
          >
            <span className={styles.msgSender}>{m.sender.email}</span>
            <span className={styles.msgBody}>{m.content}</span>
          </div>
        ))}
        <div ref={bottomRef} />
      </div>
      <div className={styles.composer}>
        <input
          value={text}
          onChange={(e) => setText(e.target.value)}
          onKeyDown={(e) => e.key === 'Enter' && send()}
          placeholder="Type a message…"
        />
        <button onClick={send}>Send</button>
      </div>
    </div>
  );
}
```
